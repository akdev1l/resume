% skill chart for the sidebar, one axis per entry in resume.json's "stats";
% defined here because the aside's \obeycr would break a multi-line tikzpicture.
\newcommand{\skillchart}{%
  \pgfmathtruncatemacro\statslast{\resumeStatsCount-1}%
  \pgfmathsetmacro\statsstep{360/\resumeStatsCount}%
  \begin{tikzpicture}[scale=0.9]
    \foreach \rad in {0.25,0.5,0.75,1}
      \draw[lightgray] (90:\rad) \foreach \i in {1,...,\statslast} { -- (90+\i*\statsstep:\rad) } -- cycle;
    \foreach \label [count=\i from 0] in \resumeStatsLabels {
      \draw[lightgray] (0,0) -- (90+\i*\statsstep:1);
      \node[font=\scriptsize, text=darkgray] at (90+\i*\statsstep:1.3) {\label};
    }
    \draw[pblue, thick, fill=pblue, fill opacity=0.25]
      \resumeStatsShape;
  \end{tikzpicture}%
}
